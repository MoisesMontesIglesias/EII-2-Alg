	private ArrayList<Integer> listadoPrimos(int n) {
	    //calcula y devuelve todos los primos hasta n
	    ArrayList<Integer> lSal= new ArrayList<>();
	    for (int i = 2; i < n+1; i++) {
	    	if (primoA1(i)) {
	        	lSal.add(i);
	        }
	    }
	    return lSal;
	}
	    		
	private boolean primoA1(int m) {
	//Devuelve si m es primo o no, mediante un algorirmo sencillo que etiquetamos como A1"""
	boolean p = true;
	for (int i = 2; i < m; i++) {
		if (m%i==0) {
			p = false;
		}
	}
		return p;
	}
